import java.util.Scanner;

public class ParityCheck {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter the string: ");
        String data = sc.nextLine();

        System.out.println("Enter the choice");
        System.out.println("1. Even Parity");
        System.out.println("2. Odd Parity");
        int choice = sc.nextInt();

        int count = 0;

        // count number of 1's
        for (int i = 0; i < data.length(); i++) {
            if (data.charAt(i) == '1') {
                count++;
            }
        }

        char parityBit;

        // Even Parity
        if (choice == 1) {
            if (count % 2 == 0)
                parityBit = '0';
            else
                parityBit = '1';
        }
        // Odd Parity
        else {
            if (count % 2 == 0)
                parityBit = '1';
            else
                parityBit = '0';
        }

        System.out.println("The string is: " + data + parityBit);
        sc.close();
    }
}

public class CharacterCount{
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        
        System.out.println("Enter the num of Frame:");
        int n=sc.nextInt();
        
        String frameString="";
        String frame="";
        
        for(int i=0;i<n;i++){
            System.out.println("Enter the frame:");
            frame=sc.next();
            int len=frame.length()+1;
            frameString=frameString+len+frame;
        }
        System.out.println("the transmitted frame:"+frameString);
        
        //receiver side
        System.out.println("\nAt Receiver:");
        System.out.println("Frame data:"+frameString);
        
        int i=0;
        System.out.println("Frame sizes are:");
        while(i<frameString.length()){
            int frameSize= Character.getNumericValue(frameString.charAt(i));
            System.out.println(frameSize+" ");
            
            String fra=frameString.substring(i+1,i+frameSize);
            System.out.println("\nFrame:"+fra);
            i+=frameSize;
        }
    }
}

import java.io.*;
import java.net.*;
import java.util.StringTokenizer;


public class MathServer {
    public static void main(String[] args) {
        try{
            //serversocket create
            ServerSocket ss=new ServerSocket(5000);
            System.out.println("Server is running...");
            
            Socket s= ss.accept();
            System.out.println("Client connected.");
            
            DataInputStream dis=new DataInputStream(s.getInputStream());
            DataOutputStream dos=new DataOutputStream(s.getOutputStream());
            
            
            String input=dis.readUTF();
            System.out.println("Received:"+input);
            
            StringTokenizer st=new StringTokenizer(input,"+-*/",true);
            int num1=Integer.parseInt(st.nextToken());
            String op=st.nextToken();
            int num2=Integer.parseInt(st.nextToken());
            
            int result=0;
            if(op.equals("+"))
                result=num1+num2;
            else if(op.equals("-"))
                result=num1-num2;
            else if(op.equals("*"))
                result=num1*num2;
            else if(op.equals("/"))
                result=num1/num2;
            
            dos.writeUTF("Result:" +result);
            
            s.close();
            ss.close();
            
            
        }catch(Exception e){
            System.out.println(e);
        }
    }
    
}

           StringTokenizer st=new StringTokenizer(data,",");

           int years=Integer.parseInt(st.nextToken());
           double loanAmount=Double.parseDouble(st.nextToken());
           double monthlyRate=annualRate/1200;
           int months=years*12;
           double monthlyPayment=(loanAmount*monthlyRate)/(1-Math.pow(1+monthlyRate, -months));
           double TotalPayment=monthlyPayment*months;
           
           dos.writeUTF("MonthlyPayment:"+monthlyPayment+"\nTotalPayment:"+TotalPayment);
           dis.close();
           dos.close();
           s.close();
           ss.close();
                                    
        }catch(Exception e){
            System.out.println(e);
        }
        
    }
  
}

import java.io.*;
import java.net.*;

public class IPClassServer {
    public static void main(String[] args) throws IOException {
        ServerSocket server = new ServerSocket(4000);
        System.out.println("Server started... Waiting for client");

        Socket socket = server.accept();
        System.out.println("Client connected.");

        DataOutputStream out = new DataOutputStream(socket.getOutputStream());

        String serverIP = InetAddress.getLocalHost().getHostAddress();
        String clientIP = socket.getInetAddress().getHostAddress();

        int firstOctet = Integer.parseInt(serverIP.split("\\.")[0]);
        String ipClass;

        if (firstOctet >= 1 && firstOctet <= 126) ipClass = "Class A";
        else if (firstOctet >= 128 && firstOctet <= 191) ipClass = "Class B";
        else if (firstOctet >= 192 && firstOctet <= 223) ipClass = "Class C";
        else if (firstOctet >= 224 && firstOctet <= 239) ipClass = "Class D";
        else ipClass = "Class E";

        System.out.println("Server IP: " + serverIP + " → " + ipClass);   
        out.writeBytes("Your IP: " + clientIP + "\nServer IP: " + serverIP + " belongs to " + ipClass + "\n");

        socket.close();
        server.close();
    }
}

    public static void main(String[] args) {
        try {
            DatagramSocket ds = new DatagramSocket(5000); // FIXED
            System.out.println("Server is running...");

            byte[] buffer = new byte[1024];
            DatagramPacket dp = new DatagramPacket(buffer, buffer.length);

            ds.receive(dp);

            String data = new String(dp.getData(), 0, dp.getLength());
            StringTokenizer st = new StringTokenizer(data, ",");

            int search = Integer.parseInt(st.nextToken());
            boolean found = false;

            while (st.hasMoreTokens()) {
                int num = Integer.parseInt(st.nextToken());
                if (num == search) {
                    found = true;
                    break;
                }
            }

            String result = found ? "Number Found" : "Number Not Found";

            byte[] sendData = result.getBytes();

            DatagramPacket reply =
                new DatagramPacket(sendData, sendData.length,
                        dp.getAddress(), dp.getPort());

            ds.send(reply);
            ds.close();

        } catch (Exception e) {
            System.out.println(e);
        }
    }
}

import java.io.*;
import java.net.*;
import java.util.StringTokenizer;

public class LargestNumberServer {
    public static void main(String[] args) throws Exception {
        // Create server socket on port 6789
        ServerSocket welcomeSocket = new ServerSocket(6789);
        System.out.println("Server is running... Waiting for client");

        while (true) {
            Socket connectionSocket = welcomeSocket.accept();
            System.out.println("Client connected!");

            // Input and output streams
            DataInputStream inFromClient = new DataInputStream(connectionSocket.getInputStream());
            DataOutputStream outToClient = new DataOutputStream(connectionSocket.getOutputStream());

            // Receive numbers as a single string
            String numbers = inFromClient.readUTF();
            System.out.println("Received from client: " + numbers);

            // Use StringTokenizer to split numbers
            StringTokenizer st = new StringTokenizer(numbers, " ");
            int largest = Integer.MIN_VALUE;

            while (st.hasMoreTokens()) {
                int num = Integer.parseInt(st.nextToken());
                if (num > largest) {
                    largest = num;
                }
            }

            // Send largest number back
            outToClient.writeUTF("Largest number is: " + largest);

            connectionSocket.close();
        }
    }
}

package Salary;
import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.util.StringTokenizer;

public class SalaryServer {
    public static void main(String[] args) {
        try {
            DatagramSocket serverSocket = new DatagramSocket(9878);
            byte[] receiveData = new byte[1024];
            byte[] sendData;

            System.out.println("Salary Server running...");

            while (true) {
                DatagramPacket receivePacket = new DatagramPacket(receiveData, receiveData.length);
                serverSocket.receive(receivePacket);

                String clientMessage = new String(receivePacket.getData(), 0, receivePacket.getLength());
                StringTokenizer st = new StringTokenizer(clientMessage, ",");

                double base = Double.parseDouble(st.nextToken());
                double houseRent = Double.parseDouble(st.nextToken());
                double incentivePercent = Double.parseDouble(st.nextToken());

                double totalSalary = base + houseRent + (base * incentivePercent / 100);
                String result = "Total Salary: " + totalSalary;

                sendData = result.getBytes();
                DatagramPacket sendPacket = new DatagramPacket(
                        sendData, sendData.length, 
                        receivePacket.getAddress(), receivePacket.getPort()
                );
                serverSocket.send(sendPacket);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
